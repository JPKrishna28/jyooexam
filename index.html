📝 C Code:
typedef struct Node {
    int data;
    struct Node* next;
} Node;

// Insert at beginning
Node* insertAtBeginning(Node* head, int data) {
    Node* newNode = (Node*)malloc(sizeof(Node));
    newNode->data = data;
    newNode->next = head;
    return newNode;
}

// Insert at end
Node* insertAtEnd(Node* head, int data) {
    Node* newNode = (Node*)malloc(sizeof(Node));
    if (head == NULL) return newNode;
    
    Node* current = head;
    while (current->next != NULL) {
        current = current->next;
    }
    current->next = newNode;
    return head;
}

// Traversal
void traverse(Node* head) {
    Node* current = head;
    while (current != NULL) {
        printf("%d -> ", current->data);
        current = current->next;
    }
    printf("NULL\n");
}

2. Doubly Linked List Operations
Operations: Creation, Insertion, Deletion, Forward & Backward Traversal
📝 C Code:
typedef struct DNode {
    int data;
    struct DNode* next;
    struct DNode* prev;
} DNode;

// Insert at beginning
DNode* insertAtBeginning(DNode* head, int data) {
    DNode* newNode = (DNode*)malloc(sizeof(DNode));
    newNode->data = data;
    if (head != NULL) {
        newNode->next = head;
        head->prev = newNode;
    }
    return newNode;
}

// Forward traversal
void traverseForward(DNode* head) {
    printf("Forward: ");
    DNode* current = head;
    while (current != NULL) {
        printf("%d <-> ", current->data);
        current = current->next;
    }
    printf("NULL\n");
}

// Backward traversal
void traverseBackward(DNode* head) {
    printf("Backward: ");
    while (head->next != NULL) head = head->next;
    while (head != NULL) {
        printf("%d <-> ", head->data);
        head = head->prev;
    }
    printf("NULL\n");
}

3. Circular Linked List Operations
Operations: Creation, Insertion, Deletion, Circular Traversal
📝 C Code:
typedef struct CNode {
    int data;
    struct CNode* next;
} CNode;

// Insert at beginning (returns new head)
CNode* insertAtBeginning(CNode* head, int data) {
    CNode* newNode = (CNode*)malloc(sizeof(CNode));
    newNode->data = data;
    
    if (head == NULL) {
        newNode->next = newNode;
        return newNode;
    }
    
    CNode* current = head;
    while (current->next != head) {
        current = current->next;
    }
    newNode->next = head;
    current->next = newNode;
    return newNode;
}

// Circular traversal
void traverse(CNode* head) {
    if (head == NULL) return;
    printf("Circular List: ");
    CNode* current = head;
    do {
        printf("%d -> ", current->data);
        current = current->next;
    } while (current != head);
    printf("(back to %d)\n", head->data);
}

4. Stack Operations
Implementations: Array-based and ADT (Linked List-based) | Operations: Push, Pop, Peek
📝 C Code (Array Implementation):
typedef struct {
    int arr[100];
    int top;
} StackArray;

void pushArray(StackArray* stack, int data) {
    if (stack->top >= 99) {
        printf("Stack Overflow!\n");
        return;
    }
    stack->arr[++stack->top] = data;
}

int popArray(StackArray* stack) {
    if (stack->top < 0) {
        printf("Stack Underflow!\n");
        return -1;
    }
    return stack->arr[stack->top--];
}
📝 C Code (ADT/Linked List Implementation):
typedef struct StackNode {
    int data;
    struct StackNode* next;
} StackNode;

void pushADT(StackADT* stack, int data) {
    StackNode* newNode = (StackNode*)malloc(sizeof(StackNode));
    newNode->data = data;
    newNode->next = stack->top;
    stack->top = newNode;
}

int popADT(StackADT* stack) {
    if (stack->top == NULL) return -1;
    StackNode* temp = stack->top;
    int data = temp->data;
    stack->top = stack->top->next;
    free(temp);
    return data;
}

5. Queue Operations
Implementations: Array-based and ADT (Linked List-based) | Operations: Enqueue, Dequeue, Peek
📝 C Code (Array Implementation):
typedef struct {
    int arr[100];
    int front, rear;
} QueueArray;

void enqueueArray(QueueArray* queue, int data) {
    if (queue->rear >= 99) {
        printf("Queue Overflow!\n");
        return;
    }
    if (queue->front == -1) queue->front = 0;
    queue->arr[++queue->rear] = data;
}

int dequeueArray(QueueArray* queue) {
    if (queue->front > queue->rear || queue->front == -1) {
        printf("Queue Underflow!\n");
        return -1;
    }
    return queue->arr[queue->front++];
}
📝 C Code (ADT/Linked List Implementation):
typedef struct QueueNode {
    int data;
    struct QueueNode* next;
} QueueNode;

void enqueueADT(QueueADT* queue, int data) {
    QueueNode* newNode = (QueueNode*)malloc(sizeof(QueueNode));
    newNode->data = data;
    newNode->next = NULL;
    
    if (queue->rear == NULL) {
        queue->front = newNode;
    } else {
        queue->rear->next = newNode;
    }
    queue->rear = newNode;
}

int dequeueADT(QueueADT* queue) {
    if (queue->front == NULL) return -1;
    QueueNode* temp = queue->front;
    int data = temp->data;
    queue->front = queue->front->next;
    if (queue->front == NULL) queue->rear = NULL;
    free(temp);
    return data;
}

void radixSort(int arr[], int n) {
    int max = getMax(arr, n);
    for (int exp = 1; max / exp > 0; exp *= 10) {
        countingSortRadix(arr, n, exp);
    }
}

📝 Heap Sort Code:
void heapSort(int arr[], int n) {
    // Build max heap
    for (int i = n / 2 - 1; i >= 0; i--) {
        heapify(arr, n, i);
    }
    // Extract elements one by one
    for (int i = n - 1; i > 0; i--) {
        // Move current root to end
        int temp = arr[0];
        arr[0] = arr[i];
        arr[i] = temp;
        heapify(arr, i, 0);
    }
}
// Time: O(n log n)

void levelOrderTraversal(TreeNode* root) {
    if (root == NULL) return;
    Queue* q = createQueue();
    enqueue(q, root);
    
    while (!isQueueEmpty(q)) {
        TreeNode* current = dequeue(q);

BSTNode* insertBST(BSTNode* root, int data) {
    if (root == NULL) return createBSTNode(data);
    
    if (data < root->data)
        root->left = insertBST(root->left, data);
    else if (data > root->data)
        root->right = insertBST(root->right, data);
    
    return root;
}
// Property: Left < Parent < Right
📝 AVL Tree Insertion with Balancing:
AVLNode* insertAVL(AVLNode* node, int data) {
    if (node == NULL) return createAVLNode(data);
    
    // Insert as in BST
    if (data < node->data)
        node->left = insertAVL(node->left, data);
    else
        node->right = insertAVL(node->right, data);
    
    // Update height
    node->height = 1 + max(getHeight(node->left), getHeight(node->right));
    
    // Get balance factor
    int balance = getBalance(node);
    
    // Left Heavy - Right Rotation
    if (balance > 1 && getBalance(node->left) >= 0)
        return rotateRight(node);
    
    // Right Heavy - Left Rotation
    if (balance < -1 && getBalance(node->right) <= 0)
        return rotateLeft(node);
}

📝 Breadth-First Search (BFS):
void bfs(Graph* graph, int start) {
    int visited[MAX] = {0};
    Queue* q = createQueue();
    
    visited[start] = 1;
    enqueue(q, start);
    
    while (!isQueueEmpty(q)) {
        int vertex = dequeue(q);
        printf("%d ", vertex);
        
        for (int i = 0; i < graph->vertices; i++) {
            if (graph->adj[vertex][i] == 1 && !visited[i]) {
                visited[i] = 1;
                enqueue(q, i);
            }
        }
    }
}
// Time: O(V + E)

Methods: Division, Multiplication, Mid-square, Folding
📝 Division Method:
// h(k) = k mod m
int divisionMethod(int key, int tableSize) {
    return key % tableSize;
}

// Simple but prone to clustering
// m should be prime for better distribution
// Example: h(23) = 23 % 100 = 23

📝 Multiplication Method:
// h(k) = floor(m * frac(k*A))
// where A is a constant (0 < A < 1)
// Often A = (√5 - 1)/2 ≈ 0.618 (golden ratio)
int multiplicationMethod(int key, int tableSize) {
    double A = 0.618033988;
    double value = (key * A);
    return (int)((value - (int)value) * tableSize);
}

// Divide key into parts and sum them
int foldingMethod(int key, int tableSize) {
    int sum = 0;
    while (key > 0) {
